cmake_minimum_required(VERSION 3.21)
project(GoldSrcStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
add_library(goldsrc_core src/core/Wad3.cpp src/core/Wad3.h)
target_link_libraries(goldsrc_core PUBLIC Qt6::Core Qt6::Gui)
target_include_directories(goldsrc_core PUBLIC src)
add_executable(GoldSrcStudio WIN32 src/main.cpp src/ui/MainWindow.cpp src/ui/MainWindow.h)
target_link_libraries(GoldSrcStudio PRIVATE goldsrc_core Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
  add_executable(wad3_tests tests/wad3_tests.cpp)
  target_link_libraries(wad3_tests PRIVATE goldsrc_core)
  add_test(NAME wad3_tests COMMAND wad3_tests)
endif()
install(TARGETS GoldSrcStudio RUNTIME DESTINATION bin)
